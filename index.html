import os import math from PIL import Image, ImageEnhance, ImageFilter import numpy as np from sklearn.cluster import KMeans class PixelArtGenerator: """ Hem görselleri hem de genel dosyaları (Binary) piksel sanatına dönüştüren AI destekli motor. """ # Hazır Renk Paletleri (RGB) PALETTES = { 'gameboy': [ [15, 56, 15], [48, 98, 48], [139, 172, 15], [155, 188, 15] ], 'cyberpunk': [ [255, 0, 122], [0, 255, 240], [255, 230, 0], [10, 10, 35], [255, 255, 255] ], 'nes_8bit': [ [0, 0, 0], [255, 255, 255], [124, 124, 124], [252, 152, 56], [0, 168, 0], [216, 0, 204], [0, 120, 248], [248, 56, 0] ] } def __init__(self): pass def apply_palette_kmeans(self, img_array, num_colors=16): """ K-Means kümeleme algoritması (AI) kullanarak resmi belirli sayıda renge indirger. """ h, w, c = img_array.shape pixels = img_array.reshape(-1, c) # Renk kümeleme kmeans = KMeans(n_clusters=num_colors, random_state=42, n_init=10) labels = kmeans.fit_predict(pixels) palette = kmeans.cluster_centers_.astype(np.uint8) # Yeni renklerle resmi tekrar oluştur quantized_pixels = palette[labels] return quantized_pixels.reshape(h, w, c) def apply_custom_palette(self, img, palette_name): """ Resmi önceden tanımlı belirli bir retro palete zorlar. """ if palette_name not in self.PALETTES: print(f"Uyarı: {palette_name} paleti bulunamadı. Orijinal renkler korunuyor.") return img palette_colors = self.PALETTES[palette_name] flat_palette = [color for rgb in palette_colors for color in rgb] # Eksik kalan renk alanlarını sıfır ile doldur (PIL palette formatı 768 eleman ister) flat_palette += [0] * (768 - len(flat_palette)) palette_img = Image.new('P', (1, 1)) palette_img.putpalette(flat_palette) return img.quantize(palette=palette_img, dither=Image.Dither.NONE).convert('RGB') def process_image(self, input_path, output_path, pixel_size=8, color_mode='kmeans', num_colors=16, custom_palette=None, outline=False): """ PNG/JPG gibi görsel dosyalarını pikselleştirir. :param pixel_size: Piksel küçültme oranı (örn: 8, 16, 32, 64). Değer büyüdükçe pikseller irileşir. :param color_mode: 'kmeans' (AI renk azaltma), 'palette' (Hazır palet) veya 'none' (Orijinal renkler). :param num_colors: AI modu seçildiyse kaç farklı renk kullanılacağı. :param custom_palette: 'gameboy', 'cyberpunk', 'nes_8bit' :param outline: Kenarlara retro siyah çizgi (kontur) eklesin mi? """ print(f"Görsel işleniyor: {input_path}") img = Image.open(input_path).convert('RGB') # Kontur Efekti (İsteğe Bağlı) if outline: edges = img.filter(ImageFilter.FIND_EDGES) edges = ImageEnhance.Contrast(edges).enhance(2.0) # 1. Aşama: Pikselleştirme (Downscaling -> Upscaling) width, height = img.size small_w = max(1, width // pixel_size) small_h = max(1, height // pixel_size) # NEAREST filtresi keskin kare pikseller oluşturur img_small = img.resize((small_w, small_h), resample=Image.Resampling.BILINEAR) # 2. Aşama: Renk Paleti ve AI İşleme if color_mode == 'kmeans': img_np = np.array(img_small) processed_np = self.apply_palette_kmeans(img_np, num_colors=num_colors) img_small = Image.fromarray(processed_np) elif color_mode == 'palette' and custom_palette: img_small = self.apply_custom_palette(img_small, custom_palette) # 3. Aşama: Orijinal Boyutuna Geri Büyütme pixel_art = img_small.resize((width, height), resample=Image.Resampling.NEAREST) # Kontur birleştirme if outline: edges_pixel = edges.resize((small_w, small_h), resample=Image.Resampling.BILINEAR) edges_pixel = edges_pixel.resize((width, height), resample=Image.Resampling.NEAREST).convert('L') # Siyah çizgileri ekle pixel_art = Image.composite(Image.new('RGB', img.size, (0, 0, 0)), pixel_art, edges_pixel) pixel_art.save(output_path) print(f"Başarıyla kaydedildi: {output_path}") def process_binary_file(self, input_path, output_path, width=256): """ Görsel OLMAYAN herhangi bir dosyayı (PDF, ZIP, EXE, TXT) piksel haritasına dönüştürür. Dosyadaki her 3 byte bir RGB pikselini temsil eder. """ print(f"Binary dosya piksel haritasına dönüştürülüyor: {input_path}") with open(input_path, 'rb') as f: data = f.read() # Veriyi 3'erli gruplar halinde RGB piksellerine böl padding = (3 - len(data) % 3) % 3 data += b'\x00' * padding # Tam bölünmesi için sıfır ekle pixels = np.frombuffer(data, dtype=np.uint8).reshape(-1, 3) total_pixels = len(pixels) height = math.ceil(total_pixels / width) # Eksik kalan pikselleri siyah ile doldur padded_pixels = np.zeros((height * width, 3), dtype=np.uint8) padded_pixels[:total_pixels] = pixels img_array = padded_pixels.reshape((height, width, 3)) img = Image.fromarray(img_array, 'RGB') img.save(output_path) print(f"Dosya piksel haritası oluşturuldu: {output_path}") # ========================================== # KULLANIM ÖRNEKLERİ VE KONTROL # ========================================== if __name__ == "__main__": ai = PixelArtGenerator() # Klasör Oluşturma os.makedirs("cikti", exist_ok=True) # --- ÖRNEK 1: Bir PNG Görselini Piksel Yapma --- # Not: Kendi resim yolunu buraya yaz ('ornek.png' yerine) input_image = "ornek.png" if os.path.exists(input_image): # 1. Çok Pikselli / Detaylı Retro (Piksel Boyutu: 4, 32 Renk) ai.process_image( input_path=input_image, output_path="cikti/piksel_hd.png", pixel_size=4, color_mode='kmeans', num_colors=32 ) # 2. Tam 8-Bit Nostalji (Piksel Boyutu: 16, GameBoy Paleti) ai.process_image( input_path=input_image, output_path="cikti/piksel_gameboy.png", pixel_size=16, color_mode='palette', custom_palette='gameboy' ) # 3. Cyberpunk Temalı Piksellendirme (Piksel Boyutu: 8, Siyah Çizgili) ai.process_image( input_path=input_image, output_path="cikti/piksel_cyberpunk.png", pixel_size=8, color_mode='palette', custom_palette='cyberpunk', outline=True ) else: print(f"'{input_image}' dosyası bulunamadı. Lütfen test etmek için klasöre bir resim koyun.") # --- ÖRNEK 2: Herhangi Bir Dosyayı Piksel Yapma (PDF, ZIP, TXT vb.) --- # Herhangi bir dosyayı piksellerden oluşan bir haritaya çevirir input_file = "belge.pdf" # Buraya istediğin dosya adını yaz if os.path.exists(input_file): ai.process_binary_file( input_path=input_file, output_path="cikti/dosya_piksel_haritasi.png", width=512 # Piksel Genişliği )
